## 🔧 核心算法详解

### 1. 深度过滤 (filter_bloom_color.fsh)
```glsl
// 线性化深度
float linearizeDepth(float depth) {
    float z = depth * 2.0 - 1.0;
    return (2.0 * DepthNear * DepthFar) / 
           (DepthFar + DepthNear - z * (DepthFar - DepthNear));
}

// 防止光晕穿透：如果深度不匹配，清除泛光
if (abs(mainDepth - diffuseDepth) > depthTolerance) {
    fragColor = vec4(0.0);
}
```

### 2. 高斯模糊 (blur.fsh)
```glsl
// 使用高斯概率密度函数
float gaussianPdf(float x, float sigma) {
    float invSigma = 1 / sigma;
    return (INV_SQRT_2PI * invSigma) * exp(-(x * x) * (0.5 * invSigma * invSigma));
}

// 双向采样模糊
for(float x = 1; x < Radius; x += 1.0) {
    float w = gaussianPdf(x, Radius);
    vec2 uvOffset = BlurDir * invSize * x;
    vec4 sample1 = texture(DiffuseSampler, texCoord + uvOffset);
    vec4 sample2 = texture(DiffuseSampler, texCoord - uvOffset);
    diffuseSum += (sample1 + sample2) * w;
}
```

### 3. Unreal 风格合成 (unreal_composite.fsh)
```glsl
// 多层Bloom混合
vec4 bloom = BloomStrength * (
    lerpBloomFactor(1.0) * texture(BlurTexture1, texCoord) +
    lerpBloomFactor(0.8) * texture(BlurTexture2, texCoord) +
    lerpBloomFactor(0.6) * texture(BlurTexture3, texCoord) +
    lerpBloomFactor(0.4) * texture(BlurTexture4, texCoord));

// 应用Jodie Reinhard色调映射
bloom.rgb = jodieReinhardTonemap(bloom.rgb);

// 根据背景亮度调整Bloom强度
float backgroundBrightness = (max + min) / 2.0;
fragColor = vec4(
    background.rgb + bloom.rgb * 
    (MinBrightness + BaseBrightness + 
     (1.0 - backgroundBrightness) * (MaxBrightness - MinBrightness)), 
    bloom.a
);
```
